\title{Purpose and limitations}
\begin{itemize}
\item
  \emA{Various matrices are solvied in TASK.}
  \begin{itemize}
  \item
    \emB{EQ,EQU}: Grad-Shafranov equation: R
  \item
    \emB{TR,TI,TX}: 1D Transport equation: R
  \item
    \emB{FP}: 2/3D Fokker-Planck equation: R
  \item
    \emB{WM,WF}: 1/2/3D fixed-frequency Maxwell's equation: C
  \item
    \emB{PIC}: Maxwell's equation: R
  \end{itemize}
\item
  \emA{Various types of band matrix solver}
  \begin{itemize}
  \item
    \emB{single process} vs \emB{MPI parallel process}
  \item
    \emB{iterative solver} vs \emB{direct solver}
  \item
    \emB{real variables} vs \emB{complex variables}
  \end{itemize}
\item
  \emA{Same motrix solver interface}
\item
  \emA{Choice of matrix solver when compile}
\item
  \emA{Limitation of single choice over TASK}
  \begin{itemize}
  \item
    \emB{single or multi},emB{iterative or direct}
  \end{itemize}
\end{itemize}

\title{Available matrix solver}
\begin{itemize}
\item
  \emA{single and direct}: band matrix solver
  \begin{itemize}
  \item
    libmtxbnd.f90, lib/libbnd.f90
  \end{itemize}
\item
  \emA{single and iterative}: PCGPME method
  \begin{itemize}
  \item
    libmtxpcg.f90, lib/libpcgpme.f90
  \end{itemize}
\item
  \emA{multi and direct}: MUMPS library
  \begin{itemize}
  \item
    libmtxmumps.f90, DMUMPS and ZMUMPS
  \end{itemize}
\item
  \emA{multi and iterative}: PETSC library
  \begin{itemize}
  \item
    libmtxksp.f90, KSP and many others
  \end{itemize}
\end{itemize}

\title{mtxp routines}

\begin{itemize}
\item
  \emA{Routine name}
  \begin{itemize}
  \item
    \emB{mtx} for real variables, 
    \emB{mtxc} for complex variables
  \end{itemize}
\item
  \emA{mtx\_initialize}: Initialize
\item
  \emA{mtx\_finalize}: Finalize
\item
  \emA{mtx\_setup}: Define matrix size
\item
  \emA{mtx\_set\_matrix}: Set matrix element
\item
  \emA{mtx\_set\_source}: Set source vector element
\item
  \emA{mtx\_set\_vector}: Set initial solution vector for iterative
  scheme
\item
  \emA{mtx\_solve}: Solve matrix equation
\item
  \emA{mtx\_get\_vector\_j}: Get $j$-th component of solution vector
\item
  \emA{mtx\_get\_vector}: Get all components of solution vector on
  single CPU
\item
  \emA{mtx\_gather\_vector}: Get all components of solution vec. for
  multi CPU
\item
  \emA{mtx\_cleanup}: Clear matrix data
\end{itemize}
  
\title{mtxp routines with arguments}

\begin{itemize}
\item
  \emB{mtx\_setup(imax,istart,iend,jwidth,nzmax,idebug)
    \begin{itemize}
    \item
      \emB{imax}: total number of matrix lines
    \item
      \emB{istart}: start of allocated range of lines
    \item
      \emB{iend}: end of allocated range of lines
    \item
      \emB{jwidth}: band matrix width
    \item
      \emB{nzmax}: number of nonzero components
    \item
      \emB{idebug}: debug level
    \end{itemize}
\item
  \emB{mtx\_set\_matrix(i,j,v)}
    \begin{itemize}
    \item
      \emB{i}: line of matrix element
    \item
      \emB{j}: row of matrix element
    \item
      \emB{v}: value of matrix element
    \end{itemize}
\item
  \emB{mtx\_set\_source(i,v)}
    \begin{itemize}
    \item
      \emB{i}: line of source vector element
    \item
      \emB{v}: value of source vector element
    \end{itemize}
\item
  \emB{mtx\_set\_vector(j,v)}: for iterative method
    \begin{itemize}
    \item
      \emB{j}: raw of initial variable vector element
    \item
      \emB{v}: value of initial variable vector element
    \end{itemize}
\item
  \emB{mtx\_solve(itype,tolerance,its,methodKSP,methodPC)j,v)}: for iterative method
    \begin{itemize}
    \item
